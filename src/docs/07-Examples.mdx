# 🎯 Examples & Use Cases

Real-world examples and patterns for using Zaman in your applications.

## 📅 Basic Examples

### Simple Date Picker

The most straightforward implementation:

```tsx
import { DatePicker } from 'zaman'

function App() {
  const handleDateChange = (date) => {
    console.log('Selected:', date.value)
  }

  return <DatePicker onChange={handleDateChange} />
}
```

### Date Picker with State

Managing selected date in component state:

```tsx
import { DatePicker } from 'zaman'
import { useState } from 'react'

function DatePickerWithState() {
  const [selectedDate, setSelectedDate] = useState<Date | null>(null)

  return (
    <div>
      <DatePicker onChange={(date) => setSelectedDate(new Date(date.value))} />

      {selectedDate && <p>You selected: {selectedDate.toLocaleDateString()}</p>}
    </div>
  )
}
```

## 📋 Form Integration

### React Hook Form

```tsx
import { useForm, Controller } from 'react-hook-form'
import { DatePicker } from 'zaman'

interface FormData {
  birthDate: string
  startDate: string
  endDate: string
}

function RegistrationForm() {
  const { control, handleSubmit } = useForm<FormData>()

  const onSubmit = (data: FormData) => {
    console.log('Form data:', data)
  }

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <div>
        <label>Birth Date</label>
        <Controller
          name="birthDate"
          control={control}
          rules={{ required: 'Birth date is required' }}
          render={({ field, fieldState }) => (
            <>
              <DatePicker
                locale="en"
                onChange={(date) => field.onChange(date.value)}
              />
              {fieldState.error && (
                <span className="error">{fieldState.error.message}</span>
              )}
            </>
          )}
        />
      </div>

      <button type="submit">Submit</button>
    </form>
  )
}
```

### Formik

```tsx
import { Formik, Form, Field } from 'formik'
import { DatePicker } from 'zaman'
import * as Yup from 'yup'

const schema = Yup.object().shape({
  eventDate: Yup.string().required('Event date is required')
})

function EventForm() {
  return (
    <Formik
      initialValues={{ eventDate: '' }}
      validationSchema={schema}
      onSubmit={(values) => console.log(values)}
    >
      {({ setFieldValue, errors, touched }) => (
        <Form>
          <Field name="eventDate">
            {() => (
              <div>
                <DatePicker
                  locale="en"
                  onChange={(date) => setFieldValue('eventDate', date.value)}
                />
                {errors.eventDate && touched.eventDate && (
                  <div className="error">{errors.eventDate}</div>
                )}
              </div>
            )}
          </Field>

          <button type="submit">Create Event</button>
        </Form>
      )}
    </Formik>
  )
}
```

## 📅 Range Picker Examples

### Vacation Booking

```tsx
import { DatePicker } from 'zaman'
import { useState } from 'react'

function VacationBooking() {
  const [range, setRange] = useState({ from: null, to: null })
  const [totalDays, setTotalDays] = useState(0)

  const handleRangeChange = ({ from, to, fromDate, toDate }) => {
    setRange({ from, to })

    if (fromDate && toDate) {
      const days = Math.ceil(
        (toDate.getTime() - fromDate.getTime()) / (1000 * 60 * 60 * 24)
      )
      setTotalDays(days)
    }
  }

  return (
    <div className="vacation-booking">
      <h2>Book Your Vacation</h2>

      <DatePicker
        range
        locale="en"
        direction="ltr"
        accentColor="#10B981"
        onChange={handleRangeChange}
      />

      {totalDays > 0 && (
        <div className="summary">
          <p>Total Days: {totalDays}</p>
          <p>From: {range.from}</p>
          <p>To: {range.to}</p>
          <button>Book Now - ${totalDays * 150}</button>
        </div>
      )}
    </div>
  )
}
```

### Date Range Filter

```tsx
import { DatePicker } from 'zaman'
import { useState, useEffect } from 'react'

function DataTable() {
  const [dateRange, setDateRange] = useState({ from: null, to: null })
  const [filteredData, setFilteredData] = useState([])

  useEffect(() => {
    if (dateRange.from && dateRange.to) {
      // Filter data based on date range
      const filtered = data.filter((item) => {
        const itemDate = new Date(item.date)
        return itemDate >= dateRange.from && itemDate <= dateRange.to
      })
      setFilteredData(filtered)
    }
  }, [dateRange])

  return (
    <div>
      <div className="filters">
        <DatePicker
          range
          locale="en"
          onChange={({ fromDate, toDate }) => {
            setDateRange({ from: fromDate, to: toDate })
          }}
        />
      </div>

      <table>{/* Render filtered data */}</table>
    </div>
  )
}
```

## ⏰ Time Picker Examples

### Alarm Clock

```tsx
import { TimePicker } from 'zaman'
import { useState } from 'react'

function AlarmClock() {
  const [alarms, setAlarms] = useState<
    Array<{
      hour: number
      minute: number
      timeConvention: string
    }>
  >([])

  const addAlarm = ({ hour, minute, timeConvention }) => {
    setAlarms([...alarms, { hour, minute, timeConvention }])
  }

  return (
    <div className="alarm-clock">
      <h2>Set Alarm</h2>

      <TimePicker
        clockTime={12}
        locale="en"
        accentColor="#EF4444"
        onChange={addAlarm}
      />

      <div className="alarms-list">
        <h3>Active Alarms</h3>
        {alarms.map((alarm, index) => (
          <div key={index} className="alarm-item">
            {alarm.hour}:{alarm.minute.toString().padStart(2, '0')}
            {alarm.timeConvention}
          </div>
        ))}
      </div>
    </div>
  )
}
```

## 🎯 Advanced Patterns

### Controlled Date Picker

```tsx
import { DatePicker } from 'zaman'
import { useState } from 'react'

function ControlledDatePicker() {
  const [isOpen, setIsOpen] = useState(false)
  const [date, setDate] = useState<Date | null>(null)

  return (
    <div>
      <button onClick={() => setIsOpen(true)}>
        {date ? date.toLocaleDateString() : 'Select Date'}
      </button>

      <DatePicker
        show={isOpen}
        defaultValue={date}
        onChange={(e) => {
          setDate(new Date(e.value))
          setIsOpen(false)
        }}
      />

      {isOpen && <button onClick={() => setIsOpen(false)}>Cancel</button>}
    </div>
  )
}
```

### Date Validation

```tsx
import { DatePicker } from 'zaman'
import { useState } from 'react'

function ValidatedDatePicker() {
  const [error, setError] = useState<string | null>(null)
  const minDate = new Date()
  minDate.setDate(minDate.getDate() + 1) // Tomorrow

  const handleChange = (date) => {
    const selectedDate = new Date(date.value)

    if (selectedDate < minDate) {
      setError('Please select a future date')
    } else {
      setError(null)
      // Process valid date
    }
  }

  return (
    <div>
      <DatePicker locale="en" onChange={handleChange} />
      {error && <span className="error">{error}</span>}
    </div>
  )
}
```

## 💾 Persistence Examples

### LocalStorage Integration

```tsx
import { DatePicker } from 'zaman'
import { useState, useEffect } from 'react'

function PersistentDatePicker() {
  const [date, setDate] = useState<Date | null>(() => {
    const saved = localStorage.getItem('selectedDate')
    return saved ? new Date(saved) : null
  })

  useEffect(() => {
    if (date) {
      localStorage.setItem('selectedDate', date.toISOString())
    }
  }, [date])

  return (
    <DatePicker
      defaultValue={date}
      locale="en"
      onChange={(d) => setDate(new Date(d.value))}
    />
  )
}
```

### URL State Sync

```tsx
import { DatePicker } from 'zaman'
import { useSearchParams } from 'react-router-dom'

function URLSyncedDatePicker() {
  const [searchParams, setSearchParams] = useSearchParams()

  const dateParam = searchParams.get('date')
  const initialDate = dateParam ? new Date(dateParam) : new Date()

  const handleChange = (date) => {
    setSearchParams({ date: date.value })
  }

  return (
    <DatePicker
      defaultValue={initialDate}
      locale="en"
      onChange={handleChange}
    />
  )
}
```

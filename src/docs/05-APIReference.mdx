# 📚 API Reference

Complete reference for all components, props, types, and utilities in Zaman.

## 📦 Components

### DatePicker

A text input with popup calendar for date selection.

#### Props

```tsx
interface DatePickerProps {
  // Value & State
  defaultValue?: Date | Dayjs | number
  show?: boolean

  // Callbacks
  onChange?: (date: DateValue) => void
  onOpen?: () => void
  onClose?: () => void

  // Range Mode
  range?: boolean
  from?: Date | Dayjs | number
  to?: Date | Dayjs | number

  // Localization
  locale?: 'fa' | 'en'
  direction?: 'rtl' | 'ltr'
  customShowDateFormat?: string

  // Styling
  accentColor?: string
  round?: 'thin' | 'x1' | 'x2' | 'x3' | 'x4'
  className?: string
  inputClass?: string

  // Calendar Config
  weekends?: number[]

  // Popup
  position?: 'left' | 'right' | 'center'

  // Input
  inputAttributes?: React.InputHTMLAttributes<HTMLInputElement>
}
```

#### Return Types

**Single Date Mode:**

```tsx
interface DateValue {
  value: string // ISO date string
  format: string // Formatted date
  date: Date // JavaScript Date object
}
```

**Range Mode:**

```tsx
interface DateRangeValue {
  from: string // ISO date string
  to: string // ISO date string
  fromDate: Date // JavaScript Date
  toDate: Date // JavaScript Date
}
```

#### Examples

```tsx
// Basic
<DatePicker onChange={(date) => console.log(date.value)} />

// With default value
<DatePicker
  defaultValue={new Date('2024-01-15')}
  onChange={(date) => console.log(date)}
/>

// Range mode
<DatePicker
  range
  from={new Date('2024-01-10')}
  to={new Date('2024-01-20')}
  onChange={({ from, to }) => console.log(from, to)}
/>

// Controlled
<DatePicker
  show={isOpen}
  onChange={(date) => {
    setDate(date)
    setIsOpen(false)
  }}
/>
```


### TimePicker

Time selection component with 12/24 hour support.

#### Props

```tsx
interface TimePickerProps {
  // Value & State
  defaultValue?: Date | Dayjs | number

  // Callback
  onChange?: (time: TimeValue) => void

  // Format
  clockTime?: 12 | 24

  // Localization
  locale?: 'fa' | 'en'

  // Styling
  accentColor?: string
  round?: 'thin' | 'x1' | 'x2' | 'x3' | 'x4'
  inputClass?: string

  // Input
  inputAttributes?: React.InputHTMLAttributes<HTMLInputElement>
}
```

#### Return Type

```tsx
interface TimeValue {
  hour: number // 0-23 (24h) or 1-12 (12h)
  minute: number // 0-59
  timeConvention?: 'AM' | 'PM' // Only in 12h mode
}
```

#### Examples

```tsx
// 24-hour format
<TimePicker
  clockTime={24}
  onChange={({ hour, minute }) => {
    console.log(`${hour}:${minute}`)
  }}
/>

// 12-hour format
<TimePicker
  clockTime={12}
  onChange={({ hour, minute, timeConvention }) => {
    console.log(`${hour}:${minute} ${timeConvention}`)
  }}
/>
```

## 🎨 Theming

### ThemeProvider

Provides theme to all child components.

#### Props

```tsx
interface ThemeProviderProps {
  theme: ThemeConfig
  children: React.ReactNode
}
```

#### Usage

```tsx
import { ThemeProvider, createTheme } from 'zaman'

const theme = createTheme({...})

<ThemeProvider theme={theme}>
  <DatePicker />
</ThemeProvider>
```

---

### createTheme

Creates a theme configuration object.

#### Signature

```tsx
function createTheme(config: DeepPartial<ThemeConfig>): ThemeConfig
```

#### ThemeConfig Type

```tsx
interface ThemeConfig {
  colors: ThemeColors

  radius: ThemeRadius
  fontSizes: ThemeFontSizes
```

### Theming Utilities

#### mergeTheme

Merges partial theme with base theme.

```tsx
function mergeTheme(
  base: ThemeConfig,
  override: DeepPartial<ThemeConfig>
): ThemeConfig
```

**Example:**

```tsx
import { defaultTheme, mergeTheme } from 'zaman'

const myTheme = mergeTheme(defaultTheme, {
  colors: {
    primary: {
      DEFAULT: '#FF6B6B'
    }
  }
})
```

#### generateAccentPalette

Generates color palette from single color.

```tsx
function generateAccentPalette(color: string): {
  DEFAULT: string
  light: string
  lighter: string
  dark: string
  foreground: string
}
```

**Example:**

```tsx
import { generateAccentPalette } from 'zaman'

const palette = generateAccentPalette('#FF6B6B')
// Returns full palette with variations
```

#### themeToCSSVariables

Converts theme object to CSS variables.

```tsx
function themeToCSSVariables(theme: ThemeConfig): Record<string, string>
```

#### applyCSSVariables

Applies CSS variables to DOM element.

```tsx
function applyCSSVariables(
  element: HTMLElement,
  variables: Record<string, string>
): void
```

## 🔍 Advanced Usage

### Using with Form Libraries

#### React Hook Form

```tsx
import { useForm, Controller } from 'react-hook-form'
import { DatePicker } from 'zaman'

function MyForm() {
  const { control, handleSubmit } = useForm()

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <Controller
        name="birthDate"
        control={control}
        render={({ field }) => (
          <DatePicker
            defaultValue={field.value}
            onChange={(date) => field.onChange(date.value)}
          />
        )}
      />
    </form>
  )
}
```

#### Formik

```tsx
import { Formik, Field } from 'formik'
import { DatePicker } from 'zaman'
;<Formik initialValues={{ date: new Date() }}>
  {({ setFieldValue, values }) => (
    <Field name="date">
      {() => (
        <DatePicker
          defaultValue={values.date}
          onChange={(date) => setFieldValue('date', date.value)}
        />
      )}
    </Field>
  )}
</Formik>
```
